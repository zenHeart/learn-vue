// tsx/Counter.tsx import { defineComponent, ref } from 'vue' export default defineComponent({ name: 'Counter', props: { initial: { type: Number, default: 0 } }, setup(props) { const count = ref(props.initial) const increment = () => count.value++ const decrement = () => count.value-- const reset = () => { count.value = 0 } // setup 返回 JSX — 等价于右侧 h() 版本 return () => ( <div class="counter"> <span>{count.value}</span> <button onClick={increment}>+1</button> <button onClick={decrement}>-1</button> <button onClick={reset}>重置</button> </div> ) } })
// 等价的 h() render 函数写法 import { h, ref } from 'vue' setup(props) { const count = ref(0) return () => h('div', { class: 'counter' }, [ h('span', {}, count.value), h('button', { onClick: () => count.value++ }, '+1'), h('button', { onClick: () => count.value-- }, '-1'), h('button', { onClick: () => { count.value = 0 }}, '重置'), ]) }
要点：setup 返回一个渲染函数，count 在 TSX 中无需 .value 自动解包，事件用 onClick 而非 @click。

// tsx/FeatureCard.tsx — 含插槽和 emit import { defineComponent, ref, useSlots } from 'vue' export default defineComponent({ name: 'FeatureCard', props: { title: { type: String, required: true } }, emits: ['update:count'], setup(props, { emit, slots }) { const count = ref(0) const s = useSlots() return () => ( <div class="fc-card"> <div class="fc-header"> <h3>{props.title}</h3> <span class="badge">{count.value}</span> </div> <div class="fc-body"> {s.default?.()} </div> <button onClick'update:count', count.value) }}>增加</button> </div> ) } }) // 使用 TSX 组件（父组件 template 中） // <FeatureCard title="特性" #default> {/* <p>插槽内容</p> {/* </FeatureCard> */} // TSX 中的插槽写法：v-slots <FeatureCard title="特性" v-slots={{ default: () => <p>插槽内容</p> }} />
要点：TSX 组件中通过 useSlots() 获取插槽，emit 用 emit('event', payload) 调用，父组件用 v-slots 传递插槽。

// Vue 3 TSX 中 v-model 需手动实现 // template: <input v-model="text" /> // TSX: setup() { const [text, setText] = useState('') return () => ( <input value={text.value} onInput={(e) => setText(e.target.value)} /> ) } // 使用 ref 的等价写法 const text = ref('') <input value={text.value} onInput={(e) => { text.value = e.target.value }} /> // 常用事件修饰符对应 // @click.stop → e.stopPropagation() // @click.prevent → e.preventDefault() // @keyup.enter → e.key === 'Enter'
要点：TSX 中没有 v-model 指令，需要绑定 value + onInput。事件修饰符 .stop/.prevent 也需手动处理。
