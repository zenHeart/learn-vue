🔄 批量更新 — 避免重复渲染

为什么需要批量更新？

// 连续修改同一个数据
this.msg = 'a';
this.msg = 'b';
this.msg = 'c';

// ❌ 无批量：触发 3 次 watcher.update() → 3 次 render
// ✅ 有批量：只触发 1 次 render（取最新值 'c'）

批量更新流程

// Vue 2 批量更新实现

let flushing = false;   // 是否正在执行队列
let waiting = false;    // 是否等待 nextTick
const queue = [];      // 待执行的 watcher 队列
const has = {};         // 去重 Set（id → true）

function queueWatcher(watcher) {
  const id = watcher.id;

  // 1️⃣ 去重：同一 watcher 只入队一次
  if (has[id]) return;
  has[id] = true;

  // 2️⃣ 入队（flushing 时倒序插入）
  if (!flushing) {
    queue.push(watcher);
  } else {
    // 正在 flush，从后往前找正确位置
    let i = queue.length - 1;
    while (i > 0 && queue[i].id > watcher.id) i--;
    queue.splice(i + 1, 0, watcher);
  }

  // 3️⃣ 触发 nextTick
  if (!waiting) {
    waiting = true;
    Promise.resolve().then(flushSchedulerQueue);
  }
}

function flushSchedulerQueue() {
  flushing = true;
  // 按 id 排序（父先子后）
  queue.sort((a, b) => a.id - b.id);

  while (queue.length) {
    const watcher = queue.shift();
    has[watcher.id] = false;
    watcher.run();  // 执行更新
  }

  flushing = false;
  waiting = false;
}

时间线对比：同步 vs 批量

同步模式 同一帧内多次修改 → 多次渲染（每次 repaint）
微任务前 watcher 入队，收集所有变更
微任务执行 flushSchedulerQueue 一次性执行队列
结果 同一 tick 内的多次修改，只触发一次 DOM 更新

nextTick 原理

// nextTick 的实现（优先使用微任务）

const callbacks = [];
let pending = false;

function nextTick(cb) {
  callbacks.push(cb);
  if (!pending) {
    pending = true;
    Promise.resolve().then(flushCallbacks);  // 微任务
  }
}

// 降级策略：
// 1. Promise.resolve() [微任务]
